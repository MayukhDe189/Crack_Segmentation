\begin{block}{1. Problem Motivation \& 60.5 : 1 Imbalance}
\small
\textbf{The Civil Engineering Challenge:} Continuous crack detection is vital for structural asphalt maintenance and pothole prevention. Cracks are ultra-thin spatial anomalies (1--5 pixels).
\begin{itemize}
    \item \textbf{Crack Pixels:} 292,885 (\textbf{1.63\%} of CFD surface area).
    \item \textbf{Pavement Pixels:} 17,710,362 (\textbf{98.37\%} of CFD surface area).
    \item \textbf{Imbalance Ratio:} \textbf{60.47 : 1} background to crack pixels.
\end{itemize}
\textbf{The Accuracy Trap:} Standard BCE is overwhelmed by the $>98\%$ background. A trivial all-background model yields 98.4\% accuracy with 0\% detection. Standard BCE thus produces broken, dotted lines (Baseline Recall was only 37.84\%).
\end{block}

\begin{block}{2. Dataset Curation \& Controlled Splits}
\small
\textbf{CrackForest Dataset (CFD):} 118 calibrated color road images with binary ground truth masks.
\begin{itemize}
    \item \textbf{Standardized Resolution:} Resized to \textbf{256 $\times$ 256} (Bilinear RGB, Nearest-Neighbor masks to preserve $\{0, 1\}$ values).
    \item \textbf{Deterministic Split (Seed = 42):}
    \begin{itemize}
        \item \textbf{Train:} 82 images (69.5\%) -- weight optimization.
        \item \textbf{Val:} 17 images (14.4\%) -- hyperparameter selection of $w$.
        \item \textbf{Test:} 19 images (16.1\%) -- held-out final benchmark.
    \end{itemize}
    \item \textbf{RAM Caching:} All 118 samples pre-cached for zero I/O latency.
\end{itemize}
\end{block}

\begin{block}{3. Symmetrical U-Net Architecture (1.94M Params)}
\small
\begin{itemize}
    \item \textbf{Encoder:} 4 stages [DoubleConv 3$\times$3 $\rightarrow$ BN $\rightarrow$ ReLU $\rightarrow$ MaxPool 2$\times$2]. Channels: $3 \rightarrow 16 \rightarrow 32 \rightarrow 64 \rightarrow 128$.
    \item \textbf{Bottleneck:} 256 channels at $16 \times 16$ spatial resolution.
    \item \textbf{Decoder:} 4 upsampling stages [ConvTranspose2d stride 2 + DoubleConv]. Channels: $256 \rightarrow 128 \rightarrow 64 \rightarrow 32 \rightarrow 16$.
    \item \textbf{Skip Connections:} Feeds high-resolution coordinates directly to decoder, preventing thin 1-pixel crack lines from vanishing.
    \item \textbf{Optimizer:} Adam ($\text{lr} = 1.5 \times 10^{-3}$, decay $= 10^{-5}$), CosineAnnealing.
\end{itemize}
\end{block}

\begin{block}{4. Loss Formulation \& Gradient Derivation}
\small
\textbf{Ordinary Binary Cross-Entropy (Baseline, $w=1.0$):}
\begin{equation*}
    \mathcal{L}_{\text{BCE}} = -\frac{1}{N} \sum_{i=1}^N \left[ y_i \log(p_i) + (1 - y_i) \log(1 - p_i) \right]
\end{equation*}
\textbf{Proposed Weighted Binary Cross-Entropy (WBCE, $w > 1.0$):}
\begin{equation*}
    \mathcal{L}_{\text{WBCE}}(w) = -\frac{1}{N} \sum_{i=1}^N \left[ w \cdot y_i \log(p_i) + (1 - y_i) \log(1 - p_i) \right]
\end{equation*}
\textbf{Gradient Derivation on Falsely Neglected Cracks:}
\begin{equation*}
    \frac{\partial \mathcal{L}_{\text{WBCE}}}{\partial z_i} = p_i - y_i \left[ 1 + (w - 1)(1 - p_i) \right]
\end{equation*}
When a true crack is missed ($y_i = 1, p_i \approx 0$), the error gradient is scaled by \textbf{$w$}, heavily penalizing false negatives. Setting $w=1.0$ recovers ordinary BCE (controlled ablation).
\end{block}

\begin{block}{9. Key Viva Defense Takeaways}
\small
\begin{itemize}
    \item \textbf{Q: Why did ordinary BCE produce broken cracks?} \\
    \textbf{A:} 98.4\% background gradient overwhelmed thin crack updates.
    \item \textbf{Q: Why not set $w = 60.5$?} \\
    \textbf{A:} Causes excessive false alarms on asphalt aggregate. Validation tuning found $w^* = 5.0$ optimal.
    \item \textbf{Q: Why are skip connections vital?} \\
    \textbf{A:} MaxPool drops 1-pixel lines; skip connections restore high-resolution coordinates directly to decoder.
\end{itemize}
